cmake_minimum_required(VERSION 3.20)
project(ThoughtlineFullscreenProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(fullscreen-selftest selftest.cpp)
if(WIN32)
 add_executable(fullscreen-probe windows.cpp)
 target_compile_definitions(fullscreen-probe PRIVATE _WIN32_WINNT=0x0A00)
 target_link_libraries(fullscreen-probe PRIVATE shell32 dwmapi user32)
elseif(APPLE)
 enable_language(OBJCXX)
 add_executable(fullscreen-probe mac.mm)
 target_compile_options(fullscreen-probe PRIVATE -fobjc-arc)
 target_link_libraries(fullscreen-probe PRIVATE "-framework AppKit" "-framework CoreGraphics" "-framework Foundation")
else()
 message(FATAL_ERROR "Native probe supports macOS and Windows only")
endif()
enable_testing()
add_test(NAME fullscreen-decisions COMMAND fullscreen-selftest)
